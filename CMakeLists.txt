cmake_minimum_required(VERSION 3.20)
project(brightwash LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
FetchContent_Declare(asio URL https://github.com/chriskohlhoff/asio/archive/refs/tags/asio-1-30-2.zip)
FetchContent_MakeAvailable(asio)
FetchContent_Declare(Crow URL https://github.com/CrowCpp/Crow/archive/refs/tags/v1.2.0.zip)
FetchContent_MakeAvailable(Crow)
add_executable(brightwash src/main.cpp)
target_include_directories(brightwash PRIVATE ${asio_SOURCE_DIR}/asio/include)
target_link_libraries(brightwash PRIVATE Crow::Crow SQLite::SQLite3 OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(brightwash PRIVATE ws2_32)
endif()
file(COPY public DESTINATION ${CMAKE_BINARY_DIR})
